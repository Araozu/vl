type Counter = object {
  value: u64,
  fun init(v: u64): *Counter {
    return Counter { value = v };
  },
  fun bump(self: *Counter): *Counter {
    return self;
  },
};

fun main() {
  var counter = Counter.init(1u64);
  counter.bump();
}
